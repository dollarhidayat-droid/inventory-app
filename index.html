/** * ==================================================================== * BACKEND APPS SCRIPT - Inventory Stok HR-GA PT Argapura * ==================================================================== * Struktur ini didesain SESUAI dengan kontrak API yang dipanggil oleh * frontend HTML (BASE_CLOUD_URL?target=xxx). * * CATATAN: Nama field/target internal (contoh: "spbList", "simpanSPB", * "eksekusiSPB", "nomorSpb") SENGAJA TIDAK diubah menjadi "FPB" supaya * kontrak data antara frontend <-> backend tetap kompatibel. Perubahan * penamaan "SPB" -> "FPB" hanya dilakukan pada teks yang tampil ke user * di frontend (label, judul, tombol, dsb). * * CARA PASANG: * 1. Buka Google Sheets baru (atau yang sudah ada). * 2. Extensions > Apps Script, hapus isi Code.gs, paste semua isi file ini. * 3. Jalankan function `setupSheets` sekali (pilih dari dropdown function, * lalu klik Run) untuk membuat semua sheet + header otomatis. * -> AMAN dijalankan ulang: sheet yang SUDAH ADA tidak akan diubah/reset, * hanya sheet baru yang akan dibuat. * 4. Deploy > New deployment > Web app. * - Execute as: Me * - Who has access: Anyone * 5. Copy URL /exec, tempel ke variabel BASE_CLOUD_URL di HTML. * ==================================================================== */ const SS = SpreadsheetApp.getActiveSpreadsheet(); // ---------- SETUP AWAL (jalankan manual sekali) ---------- function setupSheets() { getOrCreateSheet("master_data", ["barcode", "nama", "satuan", "minStock", "kategori"]); getOrCreateSheet("spb_list", ["nomorSpb", "tanggal", "department", "line", "pic", "status"]); getOrCreateSheet("spb_items", ["nomorSpb", "kode", "nama", "kategori", "qty", "satuan"]); getOrCreateSheet("transaksi", ["tanggal", "noSpb", "kode", "nama", "kategori", "line", "pic", "qty", "satuan", "jenis", "statusSpb"]); getOrCreateSheet("budget_limits", ["key", "limit"]); getOrCreateSheet("stock_awal", ["kode", "qty"]); getOrCreateSheet("users", ["username", "password", "nama", "role"]); getOrCreateSheet("adjust_history", ["timestamp", "key", "kode", "line", "limitLama", "limitBaru", "keterangan", "username", "nama"]); } function getOrCreateSheet(name, headers) { let sh = SS.getSheetByName(name); if (!sh) { sh = SS.insertSheet(name); sh.appendRow(headers); } return sh; } // ---------- ENTRY POINT ---------- function doGet(e) { try { const target = e.parameter.target; if (target === "loadAll") { return jsonOutput({ status: "SUCCESS", data: loadAllData() }); } if (target === "getAdjustHistory") { const key = e.parameter.key; return jsonOutput({ status: "SUCCESS", data: getAdjustHistory(key) }); } return jsonOutput({ status: "ERROR", message: "Target GET tidak dikenali: " + target }); } catch (err) { return jsonOutput({ status: "ERROR", message: err.toString() }); } } function doPost(e) { try { const target = e.parameter.target; const payload = e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {}; switch (target) { case "tambahUser": return jsonOutput(tambahUser(payload)); case "hapusUser": return jsonOutput(hapusUser(payload)); case "simpanSPB": return jsonOutput(simpanSPB(payload)); case "eksekusiSPB": return jsonOutput(eksekusiSPB(payload)); case "barangMasuk": return jsonOutput(barangMasuk(payload)); case "tambahMaster": return jsonOutput(tambahMaster(payload)); case "adjustBudget": return jsonOutput(adjustBudget(payload)); case "overwriteStokAktual": return jsonOutput(overwriteStokAktual(payload)); case "overwriteBudgetLineExcel": return jsonOutput(overwriteBudgetLineExcel(payload)); default: return jsonOutput({ status: "ERROR", message: "Target POST tidak dikenali: " + target }); } } catch (err) { return jsonOutput({ status: "ERROR", message: err.toString() }); } } function jsonOutput(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } // ---------- HELPER BACA SHEET ---------- function sheetToObjects(name) { const sh = SS.getSheetByName(name); if (!sh) return []; const data = sh.getDataRange().getValues(); if (data.length < 2) return []; const headers = data[0]; return data.slice(1) .filter(r => r.some(c => c !== "" && c !== null)) .map(r => { const obj = {}; headers.forEach((h, i) => obj[h] = r[i]); return obj; }); } function findRowIndexByValue(sheet, colIndex1based, value) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][colIndex1based - 1]) === String(value)) return i + 1; // 1-based row } return -1; } // ---------- LOAD ALL ---------- function loadAllData() { const masterData = sheetToObjects("master_data"); const transaksi = sheetToObjects("transaksi"); const usersRaw = sheetToObjects("users"); const spbListRaw = sheetToObjects("spb_list"); const spbItemsRaw = sheetToObjects("spb_items"); // Gabungkan item ke masing-masing FPB const spbList = spbListRaw.map(spb => { const items = spbItemsRaw.filter(it => String(it.nomorSpb) === String(spb.nomorSpb)); return { nomorSpb: spb.nomorSpb, tanggal: spb.tanggal, department: spb.department, line: spb.line, pic: spb.pic, status: spb.status, jumlahJenis: items.length, items: items.map(it => ({ kode: it.kode, nama: it.nama, kategori: it.kategori, qty: it.qty, satuan: it.satuan })) }; }); // budgetLimits jadi object map { "kode_line": limit } const budgetLimits = {}; sheetToObjects("budget_limits").forEach(r => { budgetLimits[r.key] = Number(r.limit); }); // stockAwal jadi object map { kode: qty } const stockAwal = {}; sheetToObjects("stock_awal").forEach(r => { stockAwal[r.kode] = Number(r.qty); }); return { masterData, spbList, transaksi, budgetLimits, stockAwal, users: usersRaw }; } // ---------- USERS ---------- function tambahUser(p) { const sh = SS.getSheetByName("users"); const rowIdx = findRowIndexByValue(sh, 1, p.username); if (rowIdx > -1) { sh.getRange(rowIdx, 1, 1, 4).setValues([[p.username, p.password, p.nama || "", p.role || "user"]]); } else { sh.appendRow([p.username, p.password, p.nama || "", p.role || "user"]); } return { status: "SUCCESS", message: "User disimpan" }; } function hapusUser(p) { const sh = SS.getSheetByName("users"); const rowIdx = findRowIndexByValue(sh, 1, p.username); if (rowIdx > -1) sh.deleteRow(rowIdx); return { status: "SUCCESS", message: "User dihapus" }; } // ---------- FPB (internal: spb_list / spb_items) ---------- function simpanSPB(p) { const shList = SS.getSheetByName("spb_list"); const shItems = SS.getSheetByName("spb_items"); shList.appendRow([p.nomorSpb, new Date(), p.department, p.line, p.pic, "OPEN"]); (p.items || []).forEach(it => { shItems.appendRow([p.nomorSpb, it.kode, it.nama, it.kategori, it.qty, it.satuan]); }); return { status: "SUCCESS", message: "FPB tersimpan sebagai OPEN" }; } function eksekusiSPB(p) { const shList = SS.getSheetByName("spb_list"); const rowIdx = findRowIndexByValue(shList, 1, p.nomorSpb); if (rowIdx === -1) return { status: "ERROR", message: "FPB tidak ditemukan" }; const rowData = shList.getRange(rowIdx, 1, 1, 6).getValues()[0]; const department = rowData[2], line = rowData[3], pic = rowData[4]; // Update status jadi DONE shList.getRange(rowIdx, 6).setValue("DONE"); // Catat transaksi keluar untuk setiap item FPB ini const items = sheetToObjects("spb_items").filter(it => String(it.nomorSpb) === String(p.nomorSpb)); const shTrx = SS.getSheetByName("transaksi"); const now = new Date(); items.forEach(it => { shTrx.appendRow([now, p.nomorSpb, it.kode, it.nama, it.kategori, line, pic, it.qty, it.satuan, "Keluar", "DONE"]); }); return { status: "SUCCESS", message: "FPB diselesaikan, stok terpotong" }; } // ---------- STOK MASUK (tersedia untuk user & admin di frontend) ---------- // BARU: payload sekarang bisa menyertakan { kode, qty, pic } - "pic" adalah // nama user yang sedang login saat melakukan input stok masuk, untuk // keperluan jejak/audit di Log Transaksi. function barangMasuk(p) { const master = sheetToObjects("master_data").find(m => String(m.barcode) === String(p.kode)); const shTrx = SS.getSheetByName("transaksi"); shTrx.appendRow([ new Date(), "-", p.kode, master ? master.nama : "-", master ? master.kategori : "-", "Stok Masuk", p.pic || "-", p.qty, master ? master.satuan : "-", "Masuk", "DONE" ]); return { status: "SUCCESS", message: "Barang masuk dicatat" }; } // ---------- MASTER DATA ---------- function tambahMaster(p) { const sh = SS.getSheetByName("master_data"); const rowIdx = findRowIndexByValue(sh, 1, p.barcode); if (rowIdx > -1) { sh.getRange(rowIdx, 1, 1, 5).setValues([[p.barcode, p.nama, p.satuan, p.minStock || 0, p.kategori]]); } else { sh.appendRow([p.barcode, p.nama, p.satuan, p.minStock || 0, p.kategori]); } return { status: "SUCCESS", message: "Master data disimpan" }; } // ---------- BUDGET LIMIT (JATAH MAKSIMAL BULANAN) ---------- // payload wajib kirim { kode, line, limit, keterangan, username, nama } // - key otomatis dibentuk backend = `${kode}_${line}` // - setiap perubahan otomatis dicatat ke sheet "adjust_history" function adjustBudget(p) { const sh = SS.getSheetByName("budget_limits"); const kode = p.kode || ""; const line = p.line || ""; const key = p.key || `${kode}_${line}`; const newLimit = Number(p.limit); const keterangan = p.keterangan && p.keterangan.trim() !== "" ? p.keterangan.trim() : "-"; const username = p.username || "-"; const namaUser = p.nama || "-"; const rowIdx = findRowIndexByValue(sh, 1, key); let oldLimit = 2; // default sistem jika belum pernah di-set sebelumnya if (rowIdx > -1) { oldLimit = Number(sh.getRange(rowIdx, 2).getValue()); sh.getRange(rowIdx, 2).setValue(newLimit); } else { sh.appendRow([key, newLimit]); } logAdjustHistory(key, kode, line, oldLimit, newLimit, keterangan, username, namaUser); return { status: "SUCCESS", message: "Limit budget diupdate" }; } function logAdjustHistory(key, kode, line, limitLama, limitBaru, keterangan, username, nama) { const sh = SS.getSheetByName("adjust_history"); sh.appendRow([new Date(), key, kode, line, limitLama, limitBaru, keterangan, username, nama]); } function getAdjustHistory(key) { const all = sheetToObjects("adjust_history"); return all .filter(r => String(r.key) === String(key)) .sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp)); } // ---------- OVERWRITE BATAS JATAH / MIN STOK PER ITEM PER LINE (dari upload excel) ---------- // payload: { rows: [{ kode, line, limit }, ...], username, nama } // - Menimpa sheet "budget_limits" (key = `${kode}_${line}`) secara massal dalam satu kali panggilan, // TIDAK menyentuh sheet "master_data" (kolom minStock di Master Data/Sisa Stok tetap aman). // - Setiap baris yang nilainya benar-benar berubah otomatis dicatat ke "adjust_history" supaya tetap // ada jejak audit, sama seperti perubahan manual lewat tombol Adjust. // - Baris dengan limit yang sama persis dengan nilai lama akan dilewati (tidak menulis ulang & tidak // membuat entri histori baru) supaya proses lebih cepat untuk file besar. function overwriteBudgetLineExcel(p) { const rows = p.rows || []; if (rows.length === 0) return { status: "ERROR", message: "Tidak ada baris data untuk diproses" }; const username = p.username || "-"; const namaUser = p.nama || "-"; const keteranganDefault = "Bulk Upload Excel (Sinkronisasi Batas Per Line)"; const now = new Date(); const sh = SS.getSheetByName("budget_limits"); const data = sh.getDataRange().getValues(); // [ [key, limit], ... ] termasuk header // Bangun index key -> nomor baris (1-based) supaya lookup cepat tanpa scan ulang tiap baris excel const keyToRow = {}; for (let i = 1; i < data.length; i++) { keyToRow[String(data[i][0])] = i + 1; } const historyRows = []; let updatedCount = 0; let skippedSameValue = 0; rows.forEach(r => { const kode = String(r.kode || "").trim(); const line = String(r.line || "").trim(); const newLimit = Number(r.limit); if (!kode || !line || isNaN(newLimit) || newLimit < 0) return; const key = `${kode}_${line}`; const rowIdx = keyToRow[key]; let oldLimit = 2; // default sistem jika belum pernah di-set sebelumnya if (rowIdx) { oldLimit = Number(sh.getRange(rowIdx, 2).getValue()); if (oldLimit === newLimit) { skippedSameValue++; return; } // nilai sama, tidak perlu ditulis/dicatat lagi sh.getRange(rowIdx, 2).setValue(newLimit); } else { sh.appendRow([key, newLimit]); keyToRow[key] = sh.getLastRow(); // update index supaya baris duplikat kode+line di file excel tidak dobel-append } historyRows.push([now, key, kode, line, oldLimit, newLimit, keteranganDefault, username, namaUser]); updatedCount++; }); if (historyRows.length > 0) { const shHistory = SS.getSheetByName("adjust_history"); shHistory.getRange(shHistory.getLastRow() + 1, 1, historyRows.length, 9).setValues(historyRows); } return { status: "SUCCESS", message: `${updatedCount} batas jatah per line diperbarui, ${skippedSameValue} baris dilewati (nilai sama dengan sebelumnya), dari total ${rows.length} baris terkirim.` }; } // ---------- OVERWRITE STOK AKTUAL (dari upload excel) ---------- function overwriteStokAktual(p) { const rows = p.excelData || []; const shStockAwal = SS.getSheetByName("stock_awal"); const shTrx = SS.getSheetByName("transaksi"); rows.forEach(row => { // Fleksibel terhadap nama kolom excel: barcode/kode, sisa/stok/qty const kode = row.barcode || row.kode || row.Kode || row.Barcode; const sisaBaru = row.sisa || row.stok || row.qty || row.Sisa || row.Stok || row.Qty; if (!kode || sisaBaru === undefined) return; // Hapus semua transaksi lama untuk kode ini supaya perhitungan bersih dari stock_awal baru const dataTrx = shTrx.getDataRange().getValues(); for (let i = dataTrx.length - 1; i >= 1; i--) { if (String(dataTrx[i][2]) === String(kode)) { // kolom ke-3 = kode shTrx.deleteRow(i + 1); } } // Set stock_awal = sisa aktual const rowIdx = findRowIndexByValue(shStockAwal, 1, kode); if (rowIdx > -1) { shStockAwal.getRange(rowIdx, 2).setValue(sisaBaru); } else { shStockAwal.appendRow([kode, sisaBaru]); } }); return { status: "SUCCESS", message: "Stok aktual berhasil ditimpa (overwrite)" }; }
